%===============================================================================================
% SHARED SECTION BLOCKS — the scaffolding for every section, defined ONCE.
%===============================================================================================
% The six variants differ in the ORDER of these blocks and in whether they carry a headline. The
% blocks themselves are identical across all six, so they live here rather than being pasted into
% six files where a leftmargin tweak would land in one of them and be forgotten in the other five.
%
% Each block is introduced by a section marker line. loadResumeTemplates() splits on those
% markers and substitutes the block wherever a variant file names it.
%
% THREE RULES HOLD THROUGHOUT:
%
%   1. Every list sets topsep=0pt. The gap under a section header belongs to \titlespacing and to
%      nothing else; a list that adds its own topsep double-counts it, which is how the rhythm
%      drifted section by section before. Entry separation is itemsep=\resumeEntrySep, and that
%      is the only place it is set. No block emits a bare \vspace.
%
%   2. Every \section is preceded by \resumeNeedSpace. A header that cannot fit its own first
%      content line moves to the next page WITH that content, instead of stranding itself as the
%      last line of the page — which is exactly how "Relevant Coursework" ended up alone at the
%      foot of page 1 with all nine of its courses on page 2.
%
%   3. Every bulleted list uses label=\resumebullet, which is the one round bullet defined in
%      _preamble.tex. No block picks its own marker and no block omits one.
%
% The literal \begin{itemize}[...] spelling matters: sortLatexEducationSection matches it to
% reorder education entries, and findLatexSectionBounds matches \section{Label} to drop a section
% with no data. Do not replace either with a custom environment without updating those.
%
% CONTAINS NO CANDIDATE DATA. The only non-LaTeX tokens are the uppercase placeholders, which the
% content builder replaces in a single pass.
%===============================================================================================

%%SECTION summary%%
% Plain prose, no bullet — as in all six references. leftmargin=0pt keeps it flush with the
% section rule above it.
\resumeNeedSpace{3\baselineskip}
\section{Summary}
\begin{itemize}[leftmargin=0pt, label={}, topsep=0pt, partopsep=0pt, parsep=0pt, itemsep=0pt]
\item\small{SUMMARY}
\end{itemize}

\section{Skills}

%%SECTION certifications%%
% Labelled "Certifications and Achievements", not the reference family's bare "Certifications".
% The certifications placeholder carries a MERGED, deduplicated list of the candidate's licences
% AND achievements — the two stored arrays routinely hold the same entry, and merging them is what
% stopped a HackerRank badge rendering twice. Calling that merged list "Certifications" would
% misdescribe every achievement in it, so the label says what the section actually contains.
\resumeNeedSpace{4\baselineskip}
\section{Certifications and Achievements}
\begin{itemize}[leftmargin=1.4em, label=\resumebullet, align=left, labelwidth=0.85em, labelsep=0.55em, topsep=0pt, partopsep=0pt, parsep=0pt, itemsep=\resumeBulletSep]
CERTIFICATIONS
\end{itemize}
